/**
 * Fonction principale appelée par AppSheet pour générer un audit pour un prospect.
 * @param {string} prospectKey - La clé UUID du prospect.
 * @return {string} L'URL du document d'audit généré ou un message d'erreur.
 */

// Ajouté pour faire apparaitre dans la colonne AT = RUN -> DONE > ERROR arpès l'audit de l'IA
function appsheet_generateAuditForProspect(prospectKey) {
  const FUNCTION_NAME = "appsheet_generateAuditForProspect";
  Logger.log(`🚀 ${FUNCTION_NAME} démarré pour Clé: ${prospectKey}`);

  if (!prospectKey) {
    const msg = `❌ ${FUNCTION_NAME}: prospectKey manquant`;
    Logger.log(msg);
    return msg;
  }

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(PROSPECT_SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  const rowIndexInArray = data.findIndex(r => r[COL_PROSPECT_KEY_INDEX] === prospectKey);

  if (rowIndexInArray === -1) {
    const msg = `❌ ${FUNCTION_NAME}: Prospect introuvable (${prospectKey})`;
    Logger.log(msg);
    return msg;
  }

  const row = rowIndexInArray + 1;

  try {
    // 🔄 Marquer le lancement
    sheet.getRange(row, COL_AT_INDEX + 1).setValue('RUN');

    // 🧠 Génération Audit + IA + PDF
    const auditDocUrl = generateAuditFromData(prospectKey);

    if (!auditDocUrl) {
      throw new Error("PDF non généré (URL vide)");
    }

    // ✅ SUCCÈS
    sheet.getRange(row, COL_AT_INDEX + 1).setValue('DONE');
    Logger.log(`✅ ${FUNCTION_NAME}: Audit terminé pour ${prospectKey}`);
    Logger.log(`📄 PDF : ${auditDocUrl}`);

    return auditDocUrl;

  } catch (e) {
    // ❌ ERREUR
    sheet.getRange(row, COL_AT_INDEX + 1).setValue('ERROR');
    Logger.log(`❌ ${FUNCTION_NAME} ERREUR pour ${prospectKey}`);
    Logger.log(e.message);
    Logger.log(e.stack);

    return `ERREUR SCRIPT: ${e.message}`;
  }
}

/**
 * VERSION ADAPTÉE À VOTRE LOGIQUE - ROBUSTE ET CORRIGÉE
 * @param {object} prospectDataMap Un objet avec les données du prospect.
 * @param {Array} websiteFindings Les résultats de l'audit du site web.
 * @param {string} competitionAnalysisText Le texte de l'analyse de la concurrence.
 * @param {number} nbReseaux Le nombre de réseaux sociaux calculé en temps réel.
 * @returns {object} Un objet contenant le score, les critères valides et les problèmes.
 */
function calculateAuditScore(prospectDataMap, websiteFindings, competitionAnalysisText, nbReseaux) {
    let score = 0;
    let criteresValides = [];
    let problemesDetectes = [];

    // --- Notation basée sur les données de la fiche Google ---

    // 1. Fiche Google validée
    if (prospectDataMap['Validation Fiche Google'] === "Validée") {
        score += 5;
        criteresValides.push("Fiche Google validée");
    } else {
        problemesDetectes.push("Fiche Google non validée ou statut inconnu");
    }

    // 2. Email détecté
    const email = prospectDataMap['Extrait Mail Web'] || "";
    if (email.includes("@")) {
        score += 5;
        criteresValides.push("E-mail de contact trouvé");
    } else {
        problemesDetectes.push("Aucun e-mail de contact trouvé");
    }

    // 3. Avis et Note Google
    const nbAvis = parseInt(prospectDataMap['Avis'], 10) || 0;
    const note = parseFloat(String(prospectDataMap['Note']).replace(',', '.')) || 0;
    const noteSeuilCritique = 3.5;
    const seuilAvisMinimum = 5;

    if (nbAvis >= seuilAvisMinimum) {
        if (note < noteSeuilCritique) {
            problemesDetectes.push(`❌ Note critique (${note}/5) avec ${nbAvis} avis Google`);
        } else {
            let points = 0;
            let niveau = "";
            if (nbAvis >= 30) { points = 10; niveau = (note >= 4.5) ? "⭐️⭐️⭐️⭐️⭐️ Excellente réputation" : "✅ Très bon volume d'avis"; }
            else if (nbAvis >= 20) { points = 8; }
            else if (nbAvis >= 10) { points = 6; }
            else { points = 4; }
            score += points;
            criteresValides.push((niveau || `Bonne gestion des avis`) + ` (${nbAvis} avis, ${note}/5)`);
        }
    } else {
        problemesDetectes.push(`❌ Volume d'avis insuffisant (${nbAvis} < ${seuilAvisMinimum})`);
    }

    // 4. Horaires d'ouverture (CORRIGÉ : utilise prospectDataMap)
    const horaires = prospectDataMap["Horaires d'Ouverture"] || "";
    if (horaires !== "Non renseignés" && horaires !== "Non" && horaires.length > 2) {
        score += 5;
        criteresValides.push("Horaires d'ouverture renseignés");
    } else {
        problemesDetectes.push("Horaires d'ouverture absents ou non récupérés");
    }

    // 5. Catégorie métier
    if (String(prospectDataMap['Conversion Cat']).length > 1) {
        score += 5;
        criteresValides.push("Catégorie métier principale définie");
    } else {
        problemesDetectes.push("Catégorie métier manquante");
    }

    // 6. Site Web présent
    if (String(prospectDataMap['Web Site']).startsWith("http")) {
        score += 5;
        criteresValides.push("Site Web présent");
    } else {
        problemesDetectes.push("Site Web manquant ou invalide");
    }


    // 7. Photos (Score basé sur le diagnostic secteur) Nouvelle fontion insérée le 12/01/2026
      // Mise à jour de votre fonction de Score
      const nbPhotos = parseInt(prospectDataMap['Nombre de Photos'], 10) || 0;
      const typeAPI = prospectDataMap['Catégorie ou type'] || "";
      const seuilSecteur = (configSecteurs[typeAPI] || configSecteurs['default']).seuil;

      if (nbPhotos >= seuilSecteur) { 
        score += 5; 
        criteresValides.push(`Volume photo optimal pour votre secteur (${nbPhotos}).`); 
      } else if (nbPhotos >= (seuilSecteur / 2)) { 
        score += 3; 
        criteresValides.push("Présence visuelle correcte mais améliorable."); 
      } else { 
        problemesDetectes.push(`Déficit de photos : seulement ${nbPhotos} présentes (Moyenne secteur : ${seuilSecteur}).`); 
      }

    // 8. Réseaux sociaux (CORRIGÉ : utilise le paramètre nbReseaux)
    if (nbReseaux > 2) {
        score += 10;
        criteresValides.push(`Bonne présence sur les réseaux sociaux (${nbReseaux} plateformes).`);
    } else if (nbReseaux >= 1) {
        score += 5;
        criteresValides.push(`Présence sur au moins une plateforme sociale (${nbReseaux}).`);
    } else {
        problemesDetectes.push("Aucune présence détectée sur les réseaux sociaux.");
    }

    // 9. Posts Google
    const commentairePosts = String(prospectDataMap['Commentaire Posts Google']).toLowerCase();
    if (commentairePosts.includes("posts actifs")) {
        score += 5;
        criteresValides.push("Stratégie de Posts Google active");
    } else {
        problemesDetectes.push("❌ Aucun Post Google récent");
    }

    // 10. Questions & Réponses
    const analyseQA = String(prospectDataMap['Analyse Q&A']).toLowerCase();
    if (analyseQA.includes('q&r en place')) {
        score += 5;
        criteresValides.push("Section Questions & Réponses gérée.");
    } else {
        problemesDetectes.push("Section Q&R absente ou non optimisée.");
    }

   // --- 11. Visite Virtuelle Personnalisée (VISTA 360) ---
    const visite360Pro = String(prospectDataMap['Google SV'] || "").toLowerCase();
    if (visite360Pro === "oui") {
        score += 5;
        criteresValides.push("Visite virtuelle 360° immersive (Intérieurs & Expérience client)");
    } else {
        problemesDetectes.push("Absence de visite virtuelle intérieure : vos clients ne peuvent pas 'entrer' dans l'établissement.");
    }

    // --- 12. Google Street View (La présence sur la route) ---
    const streetViewVoiture = String(prospectDataMap['Street View'] || "").toLowerCase();
    if (streetViewVoiture === "oui") {
        score += 5;
        criteresValides.push("Street View standard (Visibilité depuis la voie publique)");
    } else {
        problemesDetectes.push("Street View manquante ou obsolète : l'accès extérieur n'est pas valorisé.");
    }

    // 13. Téléphone (CORRIGÉ : utilise prospectDataMap)
    const telephone = prospectDataMap['Phone'] || "";
    if (telephone.toString().length > 5) {
        score += 5;
        criteresValides.push("Téléphone renseigné");
    } else {
        problemesDetectes.push("Téléphone manquant");
    }

    // 14. Scoring de l'audit du site web
    if (websiteFindings && websiteFindings.length > 0) {
        const goodPoints = websiteFindings.filter(f => f.startsWith('✅')).length;
        const badPoints = websiteFindings.filter(f => f.startsWith('❌')).length;
        score += goodPoints * 4;
        if (goodPoints > 0) criteresValides.push(`${goodPoints} point(s) technique(s) validé(s) sur le site web.`);
        if (badPoints > 0) problemesDetectes.push(`${badPoints} point(s) d'amélioration sur le site web.`);
    }

    // 15. Prise en compte de l'analyse concurrentielle
    if (competitionAnalysisText && competitionAnalysisText.includes("Note :")) {
        score += 5;
        criteresValides.push("Analyse concurrentielle effectuée.");
    }

   
    // 16. Apple Maps / Apple Business Connect - Niveau stratégique - 26/02/2026
    // Ajout const pour la colonne BU = Apple Status (présence manuelle)
      const appleStatus = prospectDataMap["Apple Status"] || "LEVEL_0";

      if (appleStatus === "LEVEL_2") {
        score += 10;
        criteresValides.push("🍎 Niveau 2 – Écosystème Apple maîtrisé");
      }

      else if (appleStatus === "LEVEL_1") {
        score += 5;
        criteresValides.push("🍎 Niveau 1 – Présence Apple existante");
        problemesDetectes.push("⚠️ Fiche Apple non revendiquée (contrôle partiel)");
      }

      else {
        problemesDetectes.push("🔴 Niveau 0 – Invisible dans l’écosystème Apple (Plans, Siri, iOS)");
      }         

    // Sécurité finale
    score = Math.max(0, Math.min(100, Math.round(score)));

    return { score, criteresValides, problemesDetectes };
}

/**
 * Génère une accroche commerciale à partir des problèmes détectés dans l’audit.
 *
 * @param {any} problemes - Donnée brute des problèmes détectés
 * @return {string} accroche commerciale personnalisée
 */
function generateAccroche(problemes) {
  try {
    var texte = "";

    // Gestion défensive : transformer problemes en string propre
    if (Array.isArray(problemes)) {
      texte = problemes.join(", ");
    } else if (typeof problemes === "string") {
      texte = problemes;
    } else if (problemes != null && typeof problemes.toString === "function") {
      texte = problemes.toString();
    } else {
      texte = "";
    }

    var lowerText = texte.toLowerCase();

    var accroches = [];

    if (lowerText.indexOf("aucun email") !== -1) {
      accroches.push("Je constate qu’il n’y a aucun e-mail sur votre site, ce qui peut nuire à la prise de contact rapide.");
    }
    if (lowerText.indexOf("fiche non validée") !== -1 || lowerText.indexOf("non validée") !== -1) {
      accroches.push("Votre fiche Google n’est pas validée, ce qui limite fortement votre visibilité locale.");
    }
    if (lowerText.indexOf("peu d’avis") !== -1 || lowerText.indexOf("peu d'avis") !== -1) {
      accroches.push("Vous avez peu d’avis, ce qui peut freiner la confiance des nouveaux clients.");
    }
    if (lowerText.indexOf("aucun horaire") !== -1 || lowerText.indexOf("horaires non renseignés") !== -1) {
      accroches.push("Les horaires ne sont pas renseignés, ce qui empêche vos clients de savoir quand vous êtes ouvert.");
    }
    if (accroches.length === 0) {
      accroches.push("Votre fiche Google présente des axes d’amélioration concrets. Je peux vous aider à l’optimiser !");
    }
    return accroches.join(" ");
  } catch (e) {
    Logger.log("Erreur dans generateAccroche: " + e);
    return "Accroche non disponible pour le moment.";
  }
}

/**
 * Crée un résumé textuel des critères validés et des problèmes détectés.
 * @param {string[]} criteresValides Le tableau des critères validés.
 * @param {string[]} problemesDetectes Le tableau des problèmes détectés.
 * @return {string} Un texte formaté avec des listes à puces.
 */
// Version 2 et modifié pour ajour une pharse le 05/03/2026
function generateSyntheseAudit(criteresValides, problemesDetectes) {
  let synthese = "";
  const categories = {
    "✅ Fiche & Informations": [],
    "✅ Réputation & Engagement": [],
    "✅ Contenu & Médias": [],
    "❌ Fiche & Informations": [],
    "❌ Réputation & Engagement": [],
    "❌ Contenu & Médias": []
  };

  // Classer chaque critère et problème
  const classer = (item, estCritere) => {
    const prefix = estCritere ? "✅" : "❌";
    
    // CORRECTION APPLIQUÉE ICI DANS LA CONDITION "ELSE IF"
    if (/fiche|horaires|catégorie|téléphone|site web|résumé/i.test(item)) {
        categories[`${prefix} Fiche & Informations`].push(item);
    } else if (/avis|q&r|Questions & Réponses|posts|post\b|posts actifs|Q&R en place|réseaux|sociale|sociaux|plateforme/i.test(item)) {
        categories[`${prefix} Réputation & Engagement`].push(item);
    } else if (/photos|visite|street view/i.test(item)) {
        categories[`${prefix} Contenu & Médias`].push(item);
    } else if (/apple|ios|maps/i.test(item)) {
        categories[`${prefix} Fiche & Informations`].push(item);
    }
  };
  criteresValides.forEach(c => classer(c, true));
  problemesDetectes.forEach(p => classer(p, false));

  // Construire le texte final
  for (const cat in categories) {
    if (categories[cat].length > 0) {
      synthese += `${cat}:\n`; 
      categories[cat].forEach(item => synthese += `• ${item.replace(/✅|❌/g, '').trim()}\n`); 
      synthese += ""; // \n enlevé pour tester le saut dans la synthèse
    }
  }

  // Ajout de la phrase problème détectés 
  const totalProblemes = problemesDetectes.length;

  if (totalProblemes > 0) {
    synthese += "\n" +
      "Ces " + totalProblemes +
      " éléments manquants correspondent aux signaux que Google prend en compte pour prioriser un établissement dans ses recommandations locales.";
      //" éléments manquants correspondent aux critères que Google utilise pour prioriser un établissement dans ses recommandations locales.";    
  }
   return synthese.trim();
}

/**
 * Évalue le niveau de concurrence en fonction de la catégorie de l'établissement.
 * @param {string} category La catégorie convertie du prospect (ex: "Hôtel", "Restaurant", "Photographe").
 * @return {string} Le niveau de concurrence estimé ("Faible", "Moyenne", ou "Très haute").
 */
function evaluateCompetitionLevel(category) {
  if (!category) {
    return "Moyenne"; // Une valeur par défaut si la catégorie est vide
  }

  const catLower = category.toLowerCase(); // Met en minuscules pour une comparaison robuste

  // Catégories à très forte concurrence
  const tresHaute = ["hôtel", "restaurant", "immobilier", "avocat", "garage", "coiffeur", "assurance","pharmacie"];
  if (tresHaute.some(c => catLower.includes(c))) {
    return "Très haute";
  }

  // Catégories à faible concurrence (métiers de niche, artisans spécialisés)
  const faible = ["photographe", "artiste", "consultant", "architecte", "sophrologue", "créateur"];
  if (faible.some(c => catLower.includes(c))) {
    return "Faible";
  }

  // Toutes les autres catégories sont considérées comme ayant une concurrence moyenne
  return "Moyenne";
}

/**
 * 🔹 Fonction sécurisée pour Apple Maps / Apple Business Connect
 * Retourne le niveau stratégique : LEVEL_0 / LEVEL_1 / LEVEL_2
 */
function processAppleStatus(sheet, rowIndexInArray, prospectDataMap) {
  let appleStatus = "LEVEL_0"; // Invisible

  const appleData = checkAppleMapsPresence(prospectDataMap);
  const existingABC = String(sheet.getRange(rowIndexInArray + 1, COL_BN_INDEX + 1).getValue())
                        .trim()
                        .toUpperCase();

  const appleMapsPresence = String(appleData.appleMaps).trim().toUpperCase();

  if (appleMapsPresence === "OUI") {
    appleStatus = (existingABC === "RECLAIMED") ? "LEVEL_2" : "LEVEL_1";
  }

  // Écriture Sheet
  sheet.getRange(rowIndexInArray + 1, COL_BM_INDEX + 1).setValue(appleData.appleMaps);
  sheet.getRange(rowIndexInArray + 1, COL_BO_INDEX + 1).setValue(appleData.appleMapsUrl);
  sheet.getRange(rowIndexInArray + 1, COL_BU_INDEX + 1).setValue(appleStatus);

  // 🔥 Synchronisation mémoire pour audit + AppSheet
  prospectDataMap['Apple Maps'] = appleData.appleMaps;
  prospectDataMap['Apple Business Connect (ABC)'] = existingABC || "";
  prospectDataMap['Apple Maps URL'] = appleData.appleMapsUrl;
  prospectDataMap['Apple Status'] = appleStatus;

  return appleStatus;
}

/**
 * Recherche les concurrents sur Google et extrait leur nom, note et nombre d'avis.
 * @param {string} categorie La catégorie du prospect (ex: "Restaurant italien").
 * @param {string} ville La ville du prospect (ex: "Ajaccio").
 * @returns {string} Une chaîne de texte formatée avec les résultats de l'analyse.
 */
// ========================================================= Partie 1
function analyzeCompetition(categorie, ville) {
  if (!categorie || !ville) {
    return "Analyse de la concurrence non disponible (catégorie ou ville manquante).";
  }

  const searchTerm = encodeURIComponent(`${categorie} a ${ville}`);
  const url = `https://www.google.com/search?q=${searchTerm}&hl=fr`;
  
  try {
    const htmlContent = UrlFetchApp.fetch(url, { muteHttpExceptions: true }).getContentText();
    
    // Expression régulière complexe pour trouver les concurrents dans le Local Pack
    const regex = /<div class="rllt__details">.*?<div>(.*?)<\/div>.*?<span aria-hidden="true">(.*?)<\/span>.*?<span aria-hidden="true">\((.*?)\)<\/span>/gs;
    
    let match;
    const competitors = [];
    // On boucle sur tous les résultats trouvés par la regex
    while ((match = regex.exec(htmlContent)) !== null && competitors.length < 3) {
      const name = match[1].replace(/&amp;/g, '&'); // Nom de l'établissement
      const rating = match[2].trim(); // Note (ex: "4,2")
      const reviews = match[3].trim(); // Nombre d'avis (ex: "1 234")
      
      competitors.push({ name, rating, reviews });
    }

    if (competitors.length === 0) {
      return "Aucun concurrent direct trouvé dans le Local Pack de Google pour cette recherche.";
    }

    // Formatage du résultat pour l'affichage dans le document
    let resultString = "Analyse des 3 principaux concurrents sur Google :\n\n";
    competitors.forEach(comp => {
      resultString += `▪️ **${comp.name}**\n`;
      resultString += `  Note : ${comp.rating} ⭐ (${comp.reviews} avis)\n\n`;
    });
    return resultString;

  } catch (e) {
    Logger.log(`Erreur lors de l'analyse de la concurrence : ${e.message}`);
    return "Une erreur est survenue lors de l'analyse de la concurrence.";
  }
}

/**
 * Effectue une analyse technique de base d'un site web.
 * @param {string} url L'URL du site à analyser.
 * @returns {string} Une chaîne de texte formatée avec les résultats de l'audit.
 */
function analyzeWebsite(url) {
  if (!url || !url.startsWith('http')) {
    return []; // Retourne un tableau vide si pas d'URL
  }

  let findings = [];
  try {
    // Vérification de la sécurité SSL
    if (url.startsWith('https://')) {
      findings.push("✅ Site sécurisé (utilise HTTPS)");
    } else {
      findings.push("❌ Site non sécurisé (n'utilise pas HTTPS)");
    }
    
    const htmlContent = UrlFetchApp.fetch(url, { muteHttpExceptions: true }).getContentText();
    
    // Vérification de la balise Titre
    const titleMatch = htmlContent.match(/<title>(.*?)<\/title>/i);
    if (titleMatch && titleMatch[1].trim() !== '') {
      findings.push(`✅ Balise Titre trouvée : "${titleMatch[1].trim()}"`);
    } else {
      findings.push("❌ Balise Titre manquante ou vide.");
    }
    
    // Vérification de la Méta Description
    const descMatch = htmlContent.match(/<meta\s+name="description"\s+content="(.*?)"/i);
    if (descMatch && descMatch[1].trim() !== '') {
      findings.push("✅ Méta Description trouvée.");
    } else {
      findings.push("❌ Méta Description manquante ou vide.");
    }
    
    // Vérification de l'optimisation mobile (balise viewport)
    if (htmlContent.includes('<meta name="viewport"')) {
      findings.push("✅ Site potentiellement optimisé pour les mobiles.");
    } else {
      findings.push("❌ Le site ne semble pas optimisé pour les mobiles (balise viewport manquante).");
    }
    
    return findings; // On retourne le tableau des résultats
    
  } catch (e) {
    Logger.log(`Erreur analyse site ${url}: ${e.message}`);
    return ["⚠️ Impossible d'analyser le site web."]; // Retourne un tableau avec une erreur
  }
}

/************************************************************
 *           SAFE PARSER V5 — 3 BLOCS JSON OBLIGATOIRES
 ************************************************************/
function safeParseAI_v5(raw) {
  try {
    if (!raw || typeof raw !== "string") return { error: "EMPTY_RESPONSE", raw };

    const normalized = raw.replace(/\r/g, '\n');

    const jsonBlocks = [];
    let buf = "", depth = 0, inString = false;

    // SCAN Acollades équilibrées
    for (let i = 0; i < normalized.length; i++) {
      const c = normalized[i];
      buf += c;

      if (c === '"' && normalized[i - 1] !== "\\") inString = !inString;
      if (!inString) {
        if (c === "{") depth++;
        if (c === "}") depth--;
      }

      if (depth === 0 && buf.trim().startsWith("{") && buf.trim().endsWith("}")) {
        jsonBlocks.push(buf.trim());
        buf = "";
      }
    }

    // Tentatives alternatives si < 3
    if (jsonBlocks.length < 3) {
      const matches = normalized.match(/{[\s\S]*?}/g) || [];
      const uniq = [...new Set(matches)];
      if (uniq.length >= 3) {
        jsonBlocks.length = 0;
        uniq.forEach(x => jsonBlocks.push(x));
      }
    }

    // Dernière tentative join
    if (jsonBlocks.length < 3) {
      const joined = normalized.replace(/\}\s*\n\s*\{/g, '}{');
      const matches2 = joined.match(/{[\s\S]*?}/g) || [];
      if (matches2.length >= 3) {
        jsonBlocks.length = 0;
        matches2.forEach(x => jsonBlocks.push(x));
      }
    }

    if (jsonBlocks.length < 3) {
      return { error: "JSON_BLOCK_COUNT_ERROR", found: jsonBlocks.length, raw };
    }

    const [b1, b2, b3] = jsonBlocks;

    let o1, o2, o3;
    try { o1 = JSON.parse(b1); } catch (e) { return { error: "JSON1_PARSE_ERR", block: b1 }; }
    try { o2 = JSON.parse(b2); } catch (e) { return { error: "JSON2_PARSE_ERR", block: b2 }; }
    try { o3 = JSON.parse(b3); } catch (e) { return { error: "JSON3_PARSE_ERR", block: b3 }; }

    if (!o1.result) return { error: "JSON1_SHAPE_INVALID", parsed: o1 };
    if (!Array.isArray(o2.history)) return { error: "JSON2_SHAPE_INVALID", parsed: o2 };
    if (typeof o3.score !== "number") return { error: "JSON3_SHAPE_INVALID", parsed: o3 };

    return {
      ok: true,
      json1: o1.result,
      json2: JSON.stringify(o2.history),
      json3: o3.score,
      details: { o1, o2, o3 }
    };

  } catch (err) {
    return { error: "GENERAL_PARSE_FATAL", err: err.toString(), raw };
  }
}

function callOpenAI_Vista(prompt) {

  const apiKey = PropertiesService.getScriptProperties()
    .getProperty("OPENAI_API_KEY");
  if (!apiKey) throw new Error("Clé OPENAI_API_KEY manquante");

  const url = "https://api.openai.com/v1/chat/completions";

  const payload = {
    model: "gpt-4.1-mini",
    messages: [
      { role: "system", content: "Tu es l’assistant IA de Vista360." },
      { role: "user", content: prompt }
    ],
    temperature: 0.2,
    max_tokens: 600
  };

  const res = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    headers: { Authorization: "Bearer " + apiKey },
    muteHttpExceptions: true
  });

  const rawText = res.getContentText();
  Logger.log("🧠 OpenAI RAW RESPONSE:");
  Logger.log(rawText);

  let json;
  try {
    json = JSON.parse(rawText);

  if (json?.error?.code === "insufficient_quota") {
    throw new Error("Quota OpenAI épuisé – recharge requise");
  }

  } catch (e) {
    throw new Error("JSON OpenAI non parseable");
  }

  // 🛡️ Extraction robuste
  let content = "";

  if (json?.choices?.[0]?.message?.content) {
    content = json.choices[0].message.content;
  } else if (json?.choices?.[0]?.text) {
    content = json.choices[0].text;
  } else if (json?.output_text) {
    content = json.output_text;
  } else {
    throw new Error("OpenAI a répondu sans contenu exploitable");
  }

  content = String(content || "").trim();

  if (content.length < 30) {
    throw new Error("Réponse OpenAI trop courte ou vide");
  }

  return content;
}

/**
 * Déclencheur principal de l’audit Vista360
 * AT = RUN | RUNNING | DONE | ERROR
 * AW = timestamp lancement (informatif uniquement)
 */
function onEdit(e) {
  try {
    if (!e || !e.range) return;

    const sheet = e.range.getSheet();
    if (sheet.getName() !== PROSPECT_SHEET_NAME) return;

    const row = e.range.getRow();
    const col = e.range.getColumn();

    // 👉 Colonne AT = déclencheur UNIQUE
    if (col !== COL_AT_INDEX + 1) return;

    const newValue = String(e.value || '').toUpperCase().trim();
    if (newValue !== 'RUN') return;

    // 🔒 Verrou anti double-exécution
    const lock = LockService.getScriptLock();
    if (!lock.tryLock(30000)) {
      Logger.log('⛔ Verrou actif, sortie');
      return;
    }

    try {
      // Sécurité : si déjà RUNNING, on stop
      const currentStatus = sheet.getRange(row, COL_AT_INDEX + 1).getValue();
      if (currentStatus === 'RUNNING') {
        Logger.log('⛔ Audit déjà en cours');
        return;
      }

      // Passage en RUNNING
      sheet.getRange(row, COL_AT_INDEX + 1).setValue('RUNNING');

      // Timestamp lancement (AW)
      sheet.getRange(row, COL_AW_INDEX + 1).setValue(new Date());

      // Récupération Prospect ID (UUID)
      const prospectId = sheet.getRange(row, COL_PROSPECT_KEY_INDEX + 1).getValue();
      if (!prospectId) throw new Error('Prospect ID manquant');

      Logger.log(`🚀 Lancement audit pour Prospect ID : ${prospectId}`);

      // === IA ===
      vistaAI_Process(prospectId);

      // === AUDIT + PDF ===
      generateAuditFromData(prospectId);

      // OK ne fait pas le travail de transformer AT
      //sheet.getRange(row, COL_AT_INDEX + 1).setValue('DONE');
      Logger.log('✅ Audit terminé avec succès');

    } catch (err) {
      Logger.log('❌ ERREUR AUDIT : ' + err.message);
      sheet.getRange(row, COL_AT_INDEX + 1).setValue('ERROR');
    } finally {
      lock.releaseLock();
    }

  } catch (fatal) {
    Logger.log('🔥 ERREUR FATALE onEdit : ' + fatal.message);
  }
}

      rowObj = {};
      headers.forEach((h, i) => (rowObj[h] = data[r][i]));
      break;
    }
  }

  if (!rowObj) throw new Error("Prospect introuvable.");

  const result = vistaAI_Analyse(rowObj);

  vistaAI_WriteToSheet(prospectId, result.score, result.resume);

  return result;
}

/**
 * Injecte les données IA dans le modèle Google Docs avant conversion PDF
 * Pour on utilise pas l'existant 
 */
function vistaAI_FillDoc(docId, prospectId) {
  const sh = SpreadsheetApp.getActive().getSheetByName("Prospect");
  const data = sh.getRange(2, 1, sh.getLastRow(), sh.getLastColumn()).getValues();

  const idCol = 41; // AO
  const scoreCol = 51; // AY
  const resumeCol = 52; // AZ

  let row = null;
  for (let r = 0; r < data.length; r++) {
    if (String(data[r][idCol - 1]).trim() === String(prospectId).trim()) {
      row = data[r];
      break;
    }
  }
  if (!row) throw new Error("Prospect introuvable pour injection Doc.");

  const score = row[scoreCol - 1] || "—";
  const resume = row[resumeCol - 1] || "—";

  const doc = DocumentApp.openById(docId);
  const body = doc.getBody();

  body.replaceText("{{IA_SCORE}}", String(score));
  body.replaceText("{{IA_RESUME}}", String(resume));

  doc.saveAndClose();
}

/**
 * Pipeline complet
 * 1) Duplique le modèle Docs
 * 2) Injecte Score + Résumé IA
 * 3) Convertit en PDF
 * 4) Retourne l’ID du PDF
 */
function vistaAI_GeneratePDF(modelDocId, prospectId, outputFolderId) {
  const folder = DriveApp.getFolderById(outputFolderId);
  const copy = DriveApp.getFileById(modelDocId).makeCopy("Audit_" + prospectId, folder);
  const copyId = copy.getId();

  // Injecte IA dans le doc fraîchement créé
  vistaAI_FillDoc(copyId, prospectId);

  // Conversion PDF
  const blob = DriveApp.getFileById(copyId).getBlob().getAs("application/pdf");
  const pdf = folder.createFile(blob).setName("Audit_" + prospectId + ".pdf");

  return pdf.getId();
}

/**
 * Remplace un marqueur texte par une image avec un ratio forcé 16/9
 * @param {DocumentApp.Document} doc - Le document ouvert
 * @param {string} placeholder - Le texte à chercher (ex: "{{COVER_IMAGE}}")
 * @param {string} imageUrl - L'URL de l'image
 * @param {number} width - La largeur souhaitée (ex: 500)
 */

function replaceTextWithImage(doc, placeholder, imageUrl, width) {
  const IMAGE_DEFAUT = "https://avista-360.github.io/vista360-assets/images/cover/01 COVER_IMAGE_1920x1080.jpg";
  let blob;

  // 1. Gestion du téléchargement avec sécurité
  try {
    const response = UrlFetchApp.fetch(imageUrl, { muteHttpExceptions: true });
    if (response.getResponseCode() === 200) {
      blob = response.getBlob();
    } else {
      Logger.log("⚠️ Image Google 400/403, bascule sur IMAGE_DEFAUT");
      blob = UrlFetchApp.fetch(IMAGE_DEFAUT).getBlob();
    }
  } catch (e) {
    Logger.log("❌ Erreur critique Fetch, bascule sur IMAGE_DEFAUT");
    blob = UrlFetchApp.fetch(IMAGE_DEFAUT).getBlob();
  }

  // 2. Insertion précise dans le document
  const rangeElement = doc.getBody().findText(placeholder);
  if (rangeElement) {
    const textElement = rangeElement.getElement();
    const parent = textElement.getParent();
    
    // Insertion de l'image
    let image = parent.asParagraph().appendInlineImage(blob);
    
    // Réglage strict du ratio 16/9 
    const hauteurCible = Math.round(width * (9 / 16)); 
    image.setWidth(width).setHeight(hauteurCible);
    
    // Nettoyage du marqueur {{COVER_IMAGE}}
    textElement.asText().setText("");
    
    Logger.log("✅ Image insérée proprement.");
  }
}

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(PROSPECT_SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  
  // Trouvez la première ligne avec un Place ID
  let testRow = null;
  for (let i = 1; i < data.length; i++) {
    const placeIdIndex = headers.indexOf('Place ID');
    if (data[i][placeIdIndex]) {
      testRow = i;
      break;
    }
  }
  
  if (!testRow) {
    Logger.log("  ❌ Aucun prospect avec Place ID trouvé");
    return;
  }
  
  Logger.log(`  📍 Test sur prospect ligne ${testRow + 1}`);
  const prospectDataMap = {};
  headers.forEach((header, i) => {
    prospectDataMap[header] = data[testRow][i];
  });
  
  Logger.log(`  Établissement: ${prospectDataMap['Etablissement']}`);
  Logger.log(`  Place ID: ${prospectDataMap['Place ID']}`);
  
  // Récupération sécurisée
  const placeId = prospectDataMap['Place ID'];
  if (placeId) {
    try {
      const googleDesc = getGoogleEditorialSummary(placeId);
      prospectDataMap["Google_Editorial_Summary"] = googleDesc || "Pas de description";
      Logger.log(`  ✅ Description ajoutée: ${prospectDataMap["Google_Editorial_Summary"].substring(0, 60)}...`);
    } catch (error) {
      Logger.log(`  ❌ Erreur lors de la récupération: ${error.message}`);
    }
  }
}

function testCompleteFlow() {

  // const testProspectKey = "0006532a"; // 08e167ba Avista 360

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName(PROSPECT_SHEET_NAME);
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    
    const rowIndexInArray = data.findIndex(r => r[COL_PROSPECT_KEY_INDEX] === testProspectKey);
    
    if (rowIndexInArray === -1) {
      Logger.log(`  ❌ Prospect '${testProspectKey}' non trouvé`);
      return;
    }
    
    const prospectRowDataArray = data[rowIndexInArray];
    const prospectDataMap = {};
    headers.forEach((header, i) => {
      prospectDataMap[header] = prospectRowDataArray[i];
    });
    
    Logger.log(`  📍 Prospect trouvé: ${prospectDataMap['Etablissement']}`);
    
    // === LA PARTIE À TESTER ===
    Logger.log("  ⏳ Récupération Google Editorial Summary...");
    let googleDescription = "";
    const placeId = prospectDataMap['Place ID'];
    
    if (placeId) {
      try {
        googleDescription = getGoogleEditorialSummary(placeId);
        prospectDataMap["Google_Editorial_Summary"] = googleDescription || "Pas de description fournie par Google";
        Logger.log(`  ✅ Description récupérée (${googleDescription.length} chars)`);
      } catch (error) {
        Logger.log(`  ⚠️ Erreur Google Editorial : ${error.message}`);
        prospectDataMap["Google_Editorial_Summary"] = "Erreur lors de la récupération (API Google)";
      }
    } else {
      Logger.log(`  ⚠️ Pas de Place ID`);
      prospectDataMap["Google_Editorial_Summary"] = "";
    }
    
    Logger.log(`  Final: ${prospectDataMap["Google_Editorial_Summary"]}`);
    Logger.log("  ✅ TEST RÉUSSI - Le flux ne bloque pas");
    
  } catch (error) {
    Logger.log(`  ❌ ERREUR FATALE: ${error.message}`);
  }
}

// ✅ TEST 5 : Performance (timing)
function testPerformance() {
  Logger.log("\n🧪 TEST 5 : Performance - Temps d'exécution");
  
  const placeId = "ChIJtzsXxeXg2RIRPx7CXaV5c5s";    // Site archéologique d'Aléria
  const iterations = 3;
  const times = [];
  
  for (let i = 0; i < iterations; i++) {
    const start = new Date().getTime();
    try {
      getGoogleEditorialSummary(placeId);
      const end = new Date().getTime();
      times.push(end - start);
      Logger.log(`  Itération ${i + 1}: ${end - start}ms`);
    } catch (e) {
      Logger.log(`  Itération ${i + 1}: ERREUR`);
    }
  }
  
  const avg = times.reduce((a, b) => a + b, 0) / times.length;
  Logger.log(`  ⏱️ Temps moyen: ${avg.toFixed(0)}ms`);
  
  if (avg > 5000) {
    Logger.log(`  ⚠️ ATTENTION: API lente (${avg.toFixed(0)}ms)`);
  } else {
    Logger.log(`  ✅ Performance acceptable`);
  }
}
