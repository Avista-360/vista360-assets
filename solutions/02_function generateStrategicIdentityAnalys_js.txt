function generateStrategicIdentityAnalysis(prospectDataMap, auditResult, competitionAnalysisText) {

  const payload = {
    etablissement: prospectDataMap['Etablissement'],
    ville: prospectDataMap['Ville'],
    categorie: prospectDataMap['Conversion Cat'],
    note: prospectDataMap['Note'],
    avis: prospectDataMap['Avis'],
    photos: prospectDataMap['Nombre de Photos'],
    siteWeb: prospectDataMap['Web Site'] ? "Oui" : "Non",
    appleStatus: prospectDataMap['Apple Status'],
    score: auditResult.score,
    concurrence: competitionAnalysisText
  };

  const prompt = `
Tu es un stratège en positionnement digital local.
Tu analyses l’établissement en fonction de sa catégorie d’activité.
Tu analyses l’écart entre visibilité et domination.

CONTEXTE :
${JSON.stringify(payload, null, 2)}

MISSION :
1. Tu identifies l’impact business potentiel.
2. Tu écris un texte révélateur qui déclenche une réflexion stratégique.
3. Ton style est analytique, affirmé, élégant.
4. Aucune formulation commerciale directe. Maximum 150 mots.

Style : analytique, stratégique, élégant.
Aucune formulation commerciale directe.
Ton affirmé mais respectueux.
`;

  // 1️⃣ Texte IA
  const baseText = callOpenAI(prompt);

  // 2️⃣ Calcul indice stratégique
  const strategicScore = calculateStrategicIndex(auditResult, prospectDataMap);

  const potentialLevel = strategicScore >= 7.5 ? "Élevé"
                        : strategicScore >= 6 ? "Intéressant"
                        : "Modéré";

  // 3️⃣ Bloc final injecté dans {{IDENTITE_STRATEGIQUE}}
  const finalBlock = baseText + 

  "\n\n━━━━━━━━━━━━━━━━━━\n" +
  "Indice de maturité stratégique estimé : " + strategicScore + " / 10\n" +
  "Potentiel de domination digitale : " + potentialLevel + " si structuré.";

  return finalBlock;
}

function calculateStrategicIndex(auditResult, data) {

  let base = auditResult.score / 10; // convertit score /100 vers /10

  if (data['Apple Status'] === "LEVEL_1") base -= 0.5;
  if (data['Avis'] < 50) base -= 0.3;
  if (!data['Web Site']) base -= 0.7;

  return Math.max(4.5, Math.min(9.5, base)).toFixed(1);
}
